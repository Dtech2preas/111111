import asyncio from playwright.async_api import async_playwright async def main(): async with async_playwright() as p: browser = await p.chromium.launch(headless=False) page = await browser.new_page(viewport={"width": 1280, "height": 720}) await page.goto("http://localhost:8000/editor.html") await page.wait_for_selector("#floorplan-canvas") # Select wall tool await page.click("button:has-text('Draw Wall')") # Get canvas bounding box canvas = await page.query_selector("#floorplan-canvas") box = await canvas.bounding_box() # Click on canvas to establish focus and start drawing wall start_x = box["x"] + 100 start_y = box["y"] + 100 await page.mouse.click(start_x, start_y) # Move mouse to draw end_x = start_x + 200 end_y = start_y await page.mouse.move(end_x, end_y) # Type "5" await page.keyboard.press("5") # Wait for input to be visible and have value await page.wait_for_timeout(500) # Take screenshot of input state await page.screenshot(path="wall_preview_typing3.png") await browser.close() if __name__ == "__main__": asyncio.run(main())
